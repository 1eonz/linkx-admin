LxUpload.vue · Props Bridge Protocol TypeScript Interface 
interface LxUploadProps {
  /** 允许上传的文件格式扩展名 */
  accept?: string; // 默认: ".xlsx,.csv"
  /** 单文件体积上限 (MB) */
  maxSize?: number; // 默认: 10
  /** 最大上传数量限制 */
  limit?: number; // 默认: 5
  /** 选定后是否立即自动传输 */
  autoUpload?: boolean; // 默认: false (适配表单工作流)
  /** 列表展现形式 */
  listType?: 'standard-rows' | 'compact-chips'; // 默认: 'standard-rows'
  /** 是否开启拖拽区 */
  drag?: boolean; // 默认: true (h=120px)
  /** 远端分片上传切片大小 (KB) */
  chunkSize?: number; // 默认: 1024 (1MB)
}
