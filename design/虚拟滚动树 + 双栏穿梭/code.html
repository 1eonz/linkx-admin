interface LxVirtualTreeProps<T> v2.4.0 Export 
// LxVirtualTree 虚拟树研发契约定义
export interface LxVirtualTreeProps<T = any> {
  data: T[];                       // 万级扁平或嵌套节点数组
  height?: number;                 // 容器默认 360 (px)
  itemSize?: number;               // 严格 32px 紧凑基准行高
  indent?: number;                 // 严格 16px/级
  nodeKey?: string;                // 唯一字段，默认 'id'
  showCheckbox?: boolean;          // 是否开启 Element 复选框
  checkStrictly?: boolean;         // 父子级联联动严格解耦
  filterable?: boolean;            // 是否开启快速虚拟搜索高亮
  defaultExpandedKeys?: string[];  // 默认展开节点
  scrollbarWidth?: number;         // 极细滚动条，默认 4
}
interface LxTransferPanelProps v2.4.0 Export 
// LxTransferPanel 权限双栏穿梭研发契约
export interface LxTransferPanelProps {
  treeData: Array<Record<string, any>>;  // 待选树形数据源
  modelValue: (string | number)[];      // 已选权限节点 key 列表
  titles?: [string, string];            // 默认 ['待选资源', '已授权节点']
  panelHeight?: number;                 // 默认 380px，最大 400px
  maxCount?: number;                    // 授权节点上限控制
  inheritChild?: boolean;               // 穿梭时保留下级授权开关
  onNodeChange?: (selected: any[]) => void;
  onClearAll?: () => void;              // 右侧一键清空事件
}
