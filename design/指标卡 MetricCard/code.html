// packages/components/src/LxMetricCard/types.ts
export interface LxMetricCardProps {
/** 卡片标题 (如「今日警情总数」) */
title: string;
/** 核心指标数值 (支持数字或格式化字符串，如 1,247) */
value: number | string;
/** 右侧单位 (如 '件', '人', '%', '秒') */
unit?: string;
/** 数值与进度条语义色类型：normal(主色) | success(绿) | warning(橙) | danger(红) */
status?: 'normal' | 'success' | 'warning' | 'danger'; // default: 'normal'
/** 右上角浅底小标签文案 (如「环比 +12.4%」) */
badgeText?: string;
/** 底部进度条百分比 (0 - 100)，不传则隐藏进度条 */
progress?: number;
/** 底部进度条两端标注 */
progressLabel?: string;
progressValue?: string;
}
